## C++ syntax highlighting

```cpp
#include <iostream>
#include <vector>
#include <algorithm>

template<typename T>
class BinaryTree {
    struct Node {
        T value;
        Node* left = nullptr;
        Node* right = nullptr;
        Node(T v) : value(std::move(v)) {}
    };

    Node* root = nullptr;

    void insert(Node*& node, T value) {
        if (!node) {
            node = new Node(std::move(value));
            return;
        }
        if (value < node->value)
            insert(node->left, value);
        else
            insert(node->right, value);
    }

    void inorder(Node* node, std::vector<T>& result) {
        if (!node) return;
        inorder(node->left, result);
        result.push_back(node->value);
        inorder(node->right, result);
    }

public:
    void insert(T value) { insert(root, std::move(value)); }

    std::vector<T> sorted() {
        std::vector<T> result;
        inorder(root, result);
        return result;
    }
};

int main() {
    BinaryTree<int> tree;
    for (int x : {5, 3, 7, 1, 4, 6, 8})
        tree.insert(x);

    for (int x : tree.sorted())
        std::cout << x << " ";

    return 0;
}
```

    vec2 st = gl_FragCoord.xy / u_resolution;

    float r = st.x * (0.5 + 0.5 * sin(u_time));
    float g = st.y * (0.5 + 0.5 * cos(u_time * 0.7));
    float b = 0.5 + 0.5 * sin(u_time * 0.3 + st.x * 3.14);

    gl_FragColor = vec4(r, g, b, 1.0);
}`}

void main() {
    vec2 st = gl_FragCoord.xy / u_resolution;
    vec2 center = vec2(0.5);
    float dist = distance(st, center);

        vec3(0.1, 0.2, 0.5),

    );
